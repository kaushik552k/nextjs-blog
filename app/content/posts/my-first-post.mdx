import fs from 'fs';
import path from 'path';
import matter from 'gray-matter';

// 1. Define where the files are
const rootDirectory = path.join(process.cwd(), 'src', 'content', 'posts');

export type PostMetadata = {
  slug: string;
  title: string;
  publishedAt: string;
  summary: string;
  tags?: string[];
};

// 2. Function to get all posts (for the blog listing page)
export function getBlogPosts(): PostMetadata[] {
  // Get all files in the folder
  const files = fs.readdirSync(rootDirectory);

  const posts = files.map((file) => {
    // Read the file content
    const filePath = path.join(rootDirectory, file);
    const fileContent = fs.readFileSync(filePath, 'utf-8');

    // Parse the frontmatter (metadata)
    const { data } = matter(fileContent);

    return {
      slug: file.replace('.mdx', ''),
      title: data.title,
      publishedAt: data.publishedAt,
      summary: data.summary,
      tags: data.tags,
    };
  });

  // Sort by date (newest first)
  return posts.sort((a, b) => (a.publishedAt > b.publishedAt ? -1 : 1));
}

// 3. Function to get a single post (for the actual article page)
export function getPostBySlug(slug: string) {
  const filePath = path.join(rootDirectory, `${slug}.mdx`);
  const fileContent = fs.readFileSync(filePath, 'utf-8');
  const { data, content } = matter(fileContent);

  return {
    metadata: { ...data, slug },
    content,
  };
}
